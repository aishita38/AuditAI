      {(item.permissions.can_make||item.permissions.can_check)?<form className="transfer-review-form" onSubmit={e=>{e.preventDefault();mutate(()=>reviewQueueApi.decide(item.id,item.permissions.can_make?'maker-recommend':'checker-signoff',{action,notes}));}}><label>Decision<select value={action} onChange={e=>setAction(e.target.value)}><option value="approve">Approve / explained</option><option value="reject">Reject / unresolved concern</option></select></label><p>{item.case_type==='expense_claim'?'Final sign-off changes the expense claim status.':'This decision resolves the finding only; it does not approve claims or move money.'}</p><label>Review notes<textarea required minLength={5} maxLength={5000} value={notes} onChange={e=>setNotes(e.target.value)}/></label><button className="btn btn-primary" disabled={busy||notes.trim().length<5}>{item.permissions.can_make?'Submit maker recommendation':'Submit checker sign-off'}</button></form>:<p>{item.permissions.maker_reason} {item.permissions.checker_reason}</p>}
      <form className="transfer-review-form" onSubmit={e=>{e.preventDefault();mutate(()=>reviewQueueApi.comment(item.id,comment));}}><label>Comment<textarea required minLength={5} maxLength={5000} value={comment} onChange={e=>setComment(e.target.value)}/></label><button className="btn btn-secondary" disabled={busy||comment.trim().length<5}>Add comment</button></form>
